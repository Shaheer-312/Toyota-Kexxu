In order to set:

Split it into separate commands:

bash
# Step 1: Set auto_exposure first (this makes exposure_time_absolute read-only)
sudo v4l2-ctl -d /dev/eye --set-ctrl=auto_exposure=3
# Step 2: Set everything else (skip exposure_time_absolute since auto mode controls it)
sudo v4l2-ctl -d /dev/eye \
  --set-ctrl=exposure_dynamic_framerate=0 \
  --set-ctrl=gain=0 \
  --set-ctrl=brightness=0 \
  --set-ctrl=contrast=32 \
  --set-ctrl=saturation=64 \
  --set-ctrl=hue=0 \
  --set-ctrl=gamma=100 \
  --set-ctrl=sharpness=3 \
  --set-ctrl=backlight_compensation=1 \
  --set-ctrl=white_balance_automatic=1 \
  --set-ctrl=power_line_frequency=1
# Step 3: NOW switch to manual and lock exposure
sudo v4l2-ctl -d /dev/eye --set-ctrl=auto_exposure=1
sudo v4l2-ctl -d /dev/eye --set-ctrl=exposure_time_absolute=157
The rule: Always set auto_exposure mode before trying to set exposure_time_absolute. In auto mode (3), the camera owns exposure. In manual mode (1), you own it.

Same applies for your brightness fix — set them one-by-one or in compatible groups:

bash
sudo v4l2-ctl -d /dev/eye --set-ctrl=auto_exposure=1
sudo v4l2-ctl -d /dev/eye --set-ctrl=exposure_time_absolute=500
sudo v4l2-ctl -d /dev/eye --set-ctrl=gain=100
sudo v4l2-ctl -d /dev/eye --set-ctrl=brightness=30
sudo v4l2-ctl -d /dev/eye --set-ctrl=gamma=200
sudo v4l2-ctl -d /dev/eye --set-ctrl=backlight_compensation=2
